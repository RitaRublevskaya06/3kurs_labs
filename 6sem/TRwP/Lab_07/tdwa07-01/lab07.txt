# 2. POST - добавляем новую запись

  -Method Post `
  -ContentType "application/json" `
  -Body '{"FullName":"Albert Einstein","Nationality":"DE","ReqPhotoPath":"/photos/einstein.jpg"}'

# 3. POST - добавляем вторую запись

  -Method Post `
  -ContentType "application/json" `
  -Body '{"FullName":"Marie Curie","Nationality":"PL","ReqPhotoPath":"/photos/curie.jpg"}'

# 4. GET все записи (должны быть обе)
Invoke-RestMethod -Uri http://localhost:3000/celebrities -Method Get

# 5. GET одну запись (id=1)

# 6. PUT - обновляем запись
Invoke-RestMethod -Uri http://localhost:3000/celebrities/1 `
  -Method Put `
  -ContentType "application/json" `
  -Body '{"FullName":"Albert Einstein (Updated)","Nationality":"DE","ReqPhotoPath":"/photos/einstein_new.jpg"}'

# 7. GET проверяем обновление
