// === ЗАДАНИЕ 4. ВЫБОРКА С УСЛОВНЫМИ ОПЕРАТОРАМИ ===
// Условные операторы ($lt, $gt, $or)
db.tours.find({ $or: [{ price: { $lt: 50000 } }, { duration_days: { $gt: 6 } }] }).pretty()
db.tours.find({ $or: [{ price: { $lt: 1000 } }, { duration_days: { $gt: 6 } }] }).pretty()

// Операторы работы с массивами ($all, $in)
db.clients.find({ preferences: { $all: ["пляж", "люкс"] } }).pretty()
db.tours.find({ transport: { $in: ["bus", "plane"] } }).pretty()

// Операторы $exists и $type
db.clients.find({ phone: { $exists: false } }).pretty()
db.clients.find({ phone: { $exists: false } }).pretty()

// Оператор $regex
db.clients.find({ email: { $regex: /\.ru$/ } }).pretty()



// === ЗАДАНИЕ 5. ВЫБОРКА С ПРОЕКЦИЕЙ ===
db.tours.find({}, { title: 1, price: 1, _id: 0 }).pretty()




// === ЗАДАНИЕ 6. ФУНКЦИЯ count() ===
// Число всех элементов коллекции clients
db.clients.countDocuments()

// Число элементов с фильтром (total_trips больше 3)
db.clients.countDocuments({ total_trips: { $gt: 3 } })




// === ЗАДАНИЕ 7. МЕТОДЫ limit() И skip() ===
db.tours.find().sort({ price: 1 }).skip(1).limit(1).pretty()




// === ЗАДАНИЕ 8. ФУНКЦИЯ distinct() ===
db.tours.distinct("country")
db.tours.distinct("transport")



// === ЗАДАНИЕ 9. АГРЕГАЦИОННЫЕ ОПЕРАЦИИ ===
// Группировка с суммой
db.tours.aggregate([{ $group: { _id: "$country", total_tours: { $sum: 1 } } }])

// Соединение коллекций ($lookup) и $unwind
db.clients.aggregate([
  { $unwind: "$purchases" },
  { $lookup: { from: "tours", localField: "purchases.tour_id", foreignField: "_id", as: "tour_details" } },
  { $unwind: "$tour_details" },
  { $project: { client_name: "$name", tour_title: "$tour_details.title", purchase_date: "$purchases.date", price: "$purchases.amount" } }
]).pretty()




//=== ЗАДАНИЕ 10. ПУСТОЙ И НЕПУСТОЙ FILTER, ГРУППИРВКА ===
// Пустой фильтр {} (выбирает все документы)
db.tours.aggregate([
  { $match: {} },
  { $group: { _id: "$country", total: { $sum: 1 } } }
])

// Непустой фильтр { price: { $gt: 30000 } } и группировка по нескольким ключам
db.tours.aggregate([
  { $match: { price: { $gt: 30000 } } },
  { $group: { _id: { country: "$country", stars: "$details.hotel_stars" }, count: { $sum: 1 }, avg_price: { $avg: "$price" } } },
  { $sort: { count: -1 } }
])
