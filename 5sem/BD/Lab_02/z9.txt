# 1. Заходим в контейнер
docker exec -it oracle-xe bash

# 2. Подключаемся как админ
sqlplus "sys/...@//localhost:1521/XEPDB1 as sysdba"

# 3. В SQLPlus выполняем:
ALTER USER XXXCORE IDENTIFIED BY XXX123
ALTER USER XXXCORE ACCOUNT UNLOCK;
EXIT;

# 4. Подключаемся под пользователем XXXCORE
sqlplus XXXCORE/...@//localhost:1521/XEPDB1
